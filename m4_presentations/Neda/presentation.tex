  \begin{frame}{Motivation}
    \begin{itemize}
      \setlength\itemsep{0.6em}
      \item Wien veröffentlicht \textbf{über 300 offene Datensätze} auf data.wien.gv.at
      \item Nur \textbf{8 mobile Apps} visualisieren diese Daten – meist veraltet oder spezialisiert
      \item Problem: Daten sind verfügbar, aber \textbf{nicht zugänglich}
      \item Lösung: \textbf{MapApp} – farbkodierte Pins, Offline-Download, Suche, Dunkelmodus
    \end{itemize}
  \end{frame}

  \begin{frame}{Related Work}
    \begin{itemize}
      \setlength\itemsep{0.5em}
      \item \textbf{Sayogo et al.} – 66\,\% der Open-Data-Portale bieten keine visuelle Aufbereitung
      \item \textbf{Simonofski et al.} – gut gestaltete Dashboards erhöhen Verständlichkeit; heterogene Nutzergruppen berücksichtigen
      \item \textbf{Borkin et al.} – farbige Symbole bleiben besser in Erinnerung → farbkodierte Pins
      \item \textbf{Konkurrenz}: Wien Karte, Eurostat, OECD Explorer – keine mobile Open-Data-Exploration für Wien
    \end{itemize}
  \end{frame}

  \begin{frame}{Implementation}
    \begin{itemize}
      \setlength\itemsep{0.6em}
      \item Entwickelt mit \textbf{Android Studio}, Zielgerät: Pixel 9 (API 36.1)
      \item Programmiersprache: \textbf{Kotlin} mit \textbf{Jetpack Compose}
      \item Design-System: \textbf{Material 3} – modern und barrierefrei
      \item KI-Unterstützung: \textbf{Gemini} direkt in Android Studio integriert
      \item Keine größeren technischen Herausforderungen – gegenseitige Hilfe und KI-Unterstützung
    \end{itemize}
  \end{frame}
